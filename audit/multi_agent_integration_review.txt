﻿"""Deterministic orchestration graph with bounded feedback loops.

Agent implementations are injected explicitly. This module owns execution
order, feedback control, loop accounting, and trace generation.
"""

from __future__ import annotations

from dataclasses import dataclass
import math
from typing import Any, Callable

from app.schemas import (
    COVERAGE_REPAIR_CAP,
    CRITIQUE_REVISION_CAP,
    ContractValidationError,
    LoopEvent,
    TraceEvent,
)


def _require(condition: bool, message: str) -> None:
    if not condition:
        raise ContractValidationError(message)


@dataclass(frozen=True)
class CoverageDecision:
    score: float
    accepted: bool
    reason: str
    issue: str = ""
    repairable: bool = False

    def validate(self) -> None:
        _require(
            isinstance(self.score, (int, float))
            and not isinstance(self.score, bool)
            and math.isfinite(self.score)
            and 0.0 <= self.score <= 1.0,
            "Coverage score must be finite and between 0 and 1",
        )
        _require(isinstance(self.accepted, bool), "accepted must be boolean")
        _require(
            isinstance(self.reason, str) and bool(self.reason.strip()),
            "Coverage reason must be non-blank",
        )
        _require(isinstance(self.issue, str), "Coverage issue must be a string")
        _require(isinstance(self.repairable, bool), "repairable must be boolean")
        if not self.accepted:
            _require(bool(self.issue.strip()), "Rejected coverage needs a named issue")


@dataclass(frozen=True)
class RepairProposal:
    """Explicit, caller-approved scan adjustment; never changes the source."""

    context: Any
    action: str
    reason: str
    issue: str

    def validate(self) -> None:
        for name in ("action", "reason", "issue"):
            value = getattr(self, name)
            _require(
                isinstance(value, str) and bool(value.strip()),
                f"RepairProposal.{name} must be non-blank",
            )


@dataclass(frozen=True)
class CritiqueDecision:
    passed: bool
    reason: str
    issue: str = ""

    def validate(self) -> None:
        _require(isinstance(self.passed, bool), "passed must be boolean")
        _require(
            isinstance(self.reason, str) and bool(self.reason.strip()),
            "Critique reason must be non-blank",
        )
        _require(isinstance(self.issue, str), "Critique issue must be a string")
        if not self.passed:
            _require(bool(self.issue.strip()), "Failed critique needs an issue")


@dataclass
class GraphResult:
    plan: Any
    selected_rows: Any
    aggregate: Any
    coverage: CoverageDecision
    anomalies: list[dict[str, Any]]
    diagnostics: list[dict[str, Any]]
    report: Any
    trace: list[TraceEvent]
    loop_events: list[LoopEvent]
    loop_counters: dict[str, int]


def run_execution_graph(
    *,
    request: Any,
    planner: Callable[[Any], Any],
    scanner: Callable[[Any, RepairProposal | None], Any],
    aggregator: Callable[[Any, Any], Any],
    coverage_validator: Callable[[Any, Any], CoverageDecision],
    anomaly_detector: Callable[[Any], list[dict[str, Any]]],
    diagnostic_writer: Callable[[list[dict[str, Any]]], list[dict[str, Any]]],
    critique_gate: Callable[
        [list[dict[str, Any]], list[dict[str, Any]]], CritiqueDecision
    ],
    reporter: Callable[[dict[str, Any]], Any],
    repair_provider: (
        Callable[[Any, CoverageDecision, int], RepairProposal | None] | None
    ) = None,
    diagnostic_reviser: (
        Callable[
            [list[dict[str, Any]], CritiqueDecision, list[dict[str, Any]]],
            list[dict[str, Any]],
        ]
        | None
    ) = None,
    diagnostic_fallback: (
        Callable[[list[dict[str, Any]]], list[dict[str, Any]]] | None
    ) = None,
) -> GraphResult:
    """Execute planner -> scanner -> aggregator -> validator, then analysis.

    The initial scan is not a retry. Only a repair proposal returned by the
    caller-approved repair provider can trigger a coverage retry. Diagnostic
    revisions are separately capped and never modify numeric results.
    Unexpected programming exceptions deliberately propagate.
    """
    trace: list[TraceEvent] = []
    loop_events: list[LoopEvent] = []

    def emit(
        agent: str,
        kind: str,
        msg: str,
        data: dict[str, Any],
        loop: str | None = None,
    ) -> TraceEvent:
        event = TraceEvent(
            i=len(trace),
            agent=agent,
            kind=kind,
            loop=loop,
            msg=msg,
            data=data,
        )
        event.validate()
        trace.append(event)
        return event

    plan = planner(request)
    emit(
        "planner",
        "step",
        "Execution plan created",
        {
            "action": "plan",
            "outcome": "ok",
            "reason": "Planner returned an execution plan",
        },
    )

    rows = None
    aggregate_result = None
    coverage = None
    pending_repair: RepairProposal | None = None
    repairs_used = 0
    pending_loop_event: LoopEvent | None = None
    pending_trace_event: TraceEvent | None = None

    # Initial attempt plus at most COVERAGE_REPAIR_CAP actual retries.
    for scan_number in range(COVERAGE_REPAIR_CAP + 1):
        rows = scanner(plan, pending_repair)
        emit(
            "scanner",
            "step",
            "Snapshot scan completed",
            {
                "attempt": scan_number + 1,
                "action": "scan",
                "outcome": "completed",
                "reason": (
                    "Initial scan"
                    if pending_repair is None
                    else pending_repair.reason
                ),
            },
        )

        aggregate_result = aggregator(plan, rows)
        emit(
            "aggregator",
            "step",
            "Monthly aggregation completed",
            {
                "action": "aggregate",
                "outcome": "completed",
                "reason": "Aggregator returned its result",
            },
        )

        coverage = coverage_validator(plan, aggregate_result)
        if not isinstance(coverage, CoverageDecision):
            raise ContractValidationError(
                "Coverage validator must return CoverageDecision"
            )
        coverage.validate()

        # Complete the record for the retry that led to this scan.
        if pending_loop_event is not None and pending_trace_event is not None:
            pending_loop_event.after_coverage = float(coverage.score)
            pending_loop_event.outcome = (
                "accepted" if coverage.accepted else "coverage_still_insufficient"
            )
            pending_loop_event.reason = coverage.reason

            pending_trace_event.data["after_coverage"] = float(coverage.score)
            pending_trace_event.data["outcome"] = pending_loop_event.outcome
            pending_trace_event.data["reason"] = coverage.reason

            pending_loop_event.validate()
            pending_loop_event = None
            pending_trace_event = None

        emit(
            "validator",
            "decision",
            "Coverage decision evaluated",
            {
                "action": "validate_coverage",
                "outcome": "accepted" if coverage.accepted else "not_accepted",
                "reason": coverage.reason,
                "coverage_score": float(coverage.score),
            },
        )

        if coverage.accepted:
            break

        if not coverage.repairable or repair_provider is None:
            emit(
                "validator",
                "decision",
                "Coverage repair stopped",
                {
                    "action": "stop",
                    "outcome": "repair_not_permitted",
                    "reason": (
                        "No permitted repair is available"
                        if not coverage.repairable
                        else "No repair provider was configured"
                    ),
                },
            )
            break

        if repairs_used >= COVERAGE_REPAIR_CAP:
            emit(
                "validator",
                "decision",
                "Coverage repair cap exhausted",
                {
                    "action": "stop",
                    "outcome": "max_attempts_exceeded",
                    "reason": "Coverage retry cap reached",
                    "cap": COVERAGE_REPAIR_CAP,
                },
            )
            break

        next_attempt = repairs_used + 1
        proposal = repair_provider(plan, coverage, next_attempt)

        if proposal is None:
            emit(
                "validator",
                "decision",
                "Coverage repair stopped",
                {
                    "action": "stop",
                    "outcome": "no_permitted_repair",
                    "reason": "Repair provider found no permissible adjustment",
                },
            )
            break

        if not isinstance(proposal, RepairProposal):
            raise ContractValidationError(
                "Repair provider must return RepairProposal or None"
            )
        proposal.validate()
        if not coverage.repairable:
            raise ContractValidationError(
                "Repair requested for a non-repairable coverage result"
            )

        repairs_used += 1
        pending_repair = proposal

        loop_event = LoopEvent(
            loop="coverage_repair",
            attempt=repairs_used,
            issue=coverage.issue,
            action=proposal.action,
            outcome="started",
            reason=proposal.reason,
            before_coverage=float(coverage.score),
        )
        loop_events.append(loop_event)

        loop_trace = emit(
            "validator",
            "loop",
            "Coverage feedback edge executed",
            {
                "attempt": repairs_used,
                "cap": COVERAGE_REPAIR_CAP,
                "issue": coverage.issue,
                "action": proposal.action,
                "outcome": "started",
                "reason": proposal.reason,
                "before_coverage": float(coverage.score),
                "after_coverage": None,
            },
            loop="coverage_repair",
        )
        pending_loop_event = loop_event
        pending_trace_event = loop_trace

    assert coverage is not None

    anomalies = anomaly_detector(aggregate_result)
    if not isinstance(anomalies, list):
        raise ContractValidationError("Anomaly detector must return a list")
    _require(
        all(isinstance(flag, dict) for flag in anomalies),
        "Each anomaly flag must be an object",
    )

    emit(
        "anomaly",
        "step",
        "Anomaly analysis completed",
        {
            "action": "detect_anomalies",
            "outcome": "completed",
            "reason": "Anomaly detector returned ordered flags",
            "flag_count": len(anomalies),
        },
    )

    diagnostics: list[dict[str, Any]] = []
    revision_events = 0

    # No flags means no writer or critique invocation.
    if anomalies:
        diagnostics = diagnostic_writer(anomalies)
        _require(isinstance(diagnostics, list), "Diagnostic writer must return a list")
        emit(
            "diagnostic",
            "step",
            "Initial diagnostic draft completed",
            {
                "action": "draft_diagnostics",
                "outcome": "completed",
                "reason": "Initial diagnostic draft returned",
                "diagnostic_count": len(diagnostics),
            },
        )

        critique = critique_gate(diagnostics, anomalies)
        if not isinstance(critique, CritiqueDecision):
            raise ContractValidationError(
                "Critique gate must return CritiqueDecision"
            )
        critique.validate()

        if not critique.passed and diagnostic_reviser is not None:
            for attempt in range(1, CRITIQUE_REVISION_CAP + 1):
                previous_issue = critique.issue
                diagnostics = diagnostic_reviser(
                    diagnostics, critique, anomalies
                )
                _require(
                    isinstance(diagnostics, list),
                    "Diagnostic reviser must return a list",
                )

                updated_critique = critique_gate(diagnostics, anomalies)
                if not isinstance(updated_critique, CritiqueDecision):
                    raise ContractValidationError(
                        "Critique gate must return CritiqueDecision"
                    )
                updated_critique.validate()

                revision_events += 1
                outcome = (
                    "accepted"
                    if updated_critique.passed
                    else "revision_still_required"
                )
                loop_event = LoopEvent(
                    loop="critique_revision",
                    attempt=attempt,
                    issue=previous_issue,
                    action="revise_diagnostics",
                    outcome=outcome,
                    reason=updated_critique.reason,
                )
                loop_events.append(loop_event)

                emit(
                    "diagnostic",
                    "loop",
                    "Diagnostic critique/revision executed",
                    {
                        "attempt": attempt,
                        "cap": CRITIQUE_REVISION_CAP,
                        "issue": previous_issue,
                        "action": "revise_diagnostics",
                        "outcome": outcome,
                        "reason": updated_critique.reason,
                    },
                    loop="critique_revision",
                )

                critique = updated_critique
                if critique.passed:
                    break

        if not critique.passed:
            if diagnostic_fallback is None:
                raise ContractValidationError(
                    "Critique failed and no deterministic diagnostic fallback exists"
                )
            diagnostics = diagnostic_fallback(anomalies)
            _require(
                isinstance(diagnostics, list),
                "Diagnostic fallback must return a list",
            )
            emit(
                "diagnostic",
                "warning",
                "Deterministic diagnostic fallback used",
                {
                    "action": "fallback",
                    "outcome": "template",
                    "reason": (
                        "Critique did not pass within the configured revision policy"
                    ),
                },
            )

    loop_counters = {
        "coverage_repair": sum(
            event.loop == "coverage_repair" for event in loop_events
        ),
        "critique_revision": sum(
            event.loop == "critique_revision" for event in loop_events
        ),
    }

    for name, cap in (
        ("coverage_repair", COVERAGE_REPAIR_CAP),
        ("critique_revision", CRITIQUE_REVISION_CAP),
    ):
        _require(
            loop_counters[name] <= cap,
            f"{name} loop exceeded its configured cap",
        )

    for event in trace:
        event.validate()

    trace_counts = {
        name: sum(
            event.kind == "loop" and event.loop == name for event in trace
        )
        for name in loop_counters
    }
    _require(
        trace_counts == loop_counters,
        "Loop counters do not match loop trace events",
    )

    status = "ok" if coverage.accepted else "partial"
    coverage_decision = (
        "accept"
        if coverage.accepted
        else "partial"
        if coverage.score > 0
        else "fail"
    )

    # Log reporter entry before assembling the payload, so the final trace
    # contains a visible report-assembly event.
    emit(
        "reporter",
        "step",
        "Final report assembly started",
        {
            "action": "assemble_report",
            "outcome": "started",
            "reason": "Validated outputs and execution trace are ready",
        },
    )

    payload = {
        "plan": plan,
        "selected_rows": rows,
        "aggregate": aggregate_result,
        "coverage": {
            "score": float(coverage.score),
            "accepted": coverage.accepted,
            "decision": coverage_decision,
            "reason": coverage.reason,
        },
        "anomalies": anomalies,
        "diagnostics": diagnostics,
        "status": status,
        "loop_counters": dict(loop_counters),
        "loop_events": [event.to_dict() for event in loop_events],
        "trace": [event.to_dict() for event in trace],
    }

    report = reporter(payload)

    return GraphResult(
        plan=plan,
        selected_rows=rows,
        aggregate=aggregate_result,
        coverage=coverage,
        anomalies=anomalies,
        diagnostics=diagnostics,
        report=report,
        trace=trace,
        loop_events=loop_events,
        loop_counters=loop_counters,
    )

"""Build and validate the runtime state from the frozen Hyderabad CSV."""

from __future__ import annotations

import hashlib
import json
import sys
from pathlib import Path

import numpy as np
import pandas as pd

ROOT = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(ROOT))

from app.schemas import (  # noqa: E402
    AnalysisRequest,
    CoverageFlag,
    MonthState,
    RuntimeState,
    RunPlan,
    RetryCounters,
    SourcePeriod,
    EXPECTED_MONTHS,
    CONTRACT_VERSION,
)

RAW_FILE = ROOT / "data" / "raw" / "hyderabad.csv"
OUTPUT_FILE = ROOT / "audit" / "runtime_state.json"

EXPECTED_SHA256 = (
    "9e81698322606c86b4a23e41ace5deecff5e5ed94dc768e9e5e371a2bc48898a"
)
PRICE_SENTINEL = 2147483647


def sha256_file(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for chunk in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(chunk)
    return digest.hexdigest()


def numeric_column(frame: pd.DataFrame, name: str) -> pd.Series:
    return pd.to_numeric(frame[name], errors="coerce")


def main() -> None:
    if not RAW_FILE.is_file():
        raise FileNotFoundError(f"Raw CSV not found: {RAW_FILE}")

    source_hash = sha256_file(RAW_FILE)
    if source_hash != EXPECTED_SHA256:
        raise RuntimeError(
            "snapshot_integrity_error: the CSV hash does not match "
            "the frozen CONTRACT.md / Golden.md snapshot. "
            f"Expected {EXPECTED_SHA256}; got {source_hash}"
        )

    df = pd.read_csv(RAW_FILE, low_memory=False)
    raw_rows, raw_columns = df.shape

    required = {
        "POSTING_DATE",
        "PREFERENCE",
        "MIN_PRICE",
        "MAX_PRICE",
        "PRICE_SQFT",
        "PROP_ID",
    }
    missing_columns = sorted(required - set(df.columns))
    if missing_columns:
        raise ValueError(f"Missing required CSV columns: {missing_columns}")

    # POSTING_DATE is Unix milliseconds. Localize the instant to UTC,
    # convert to Asia/Kolkata, then derive the month.
    raw_timestamp = numeric_column(df, "POSTING_DATE")
    valid_timestamp_number = (
        raw_timestamp.notna()
        & np.isfinite(raw_timestamp)
    )

    posting_date = pd.Series(
        pd.NaT,
        index=df.index,
        dtype="datetime64[ns, Asia/Kolkata]",
    )

    parsed_utc = pd.to_datetime(
        raw_timestamp.loc[valid_timestamp_number],
        unit="ms",
        utc=True,
        errors="coerce",
    )
    posting_date.loc[parsed_utc.index] = parsed_utc.dt.tz_convert(
        "Asia/Kolkata"
    )

    df["_posting_date_local"] = posting_date
    df["_month"] = df["_posting_date_local"].dt.strftime("%Y-%m")

    valid_date_mask = df["_posting_date_local"].notna()
    in_window_mask = (
        valid_date_mask
        & df["_month"].isin(EXPECTED_MONTHS)
    )

    in_window = df.loc[in_window_mask].copy()

    # Normalize only the documented sale indicator.
    preference = in_window["PREFERENCE"].astype("string").str.strip().str.upper()
    sale_mask = preference.eq("S").fillna(False)
    in_window["_is_sale"] = sale_mask

    min_price = numeric_column(in_window, "MIN_PRICE")
    max_price = numeric_column(in_window, "MAX_PRICE")

    valid_min = (
        min_price.notna()
        & np.isfinite(min_price)
        & (min_price > 0)
        & (min_price != PRICE_SENTINEL)
    )
    valid_max = (
        max_price.notna()
        & np.isfinite(max_price)
        & (max_price > 0)
        & (max_price != PRICE_SENTINEL)
    )

    coherent_range = (
        in_window["_is_sale"]
        & valid_min
        & valid_max
        & (min_price <= max_price)
    )

    in_window["_min_price_numeric"] = min_price
    in_window["_max_price_numeric"] = max_price
    in_window["_valid_price_range"] = coherent_range

    price_sqft = numeric_column(in_window, "PRICE_SQFT")
    valid_price_sqft = (
        in_window["_is_sale"]
        & price_sqft.notna()
        & np.isfinite(price_sqft)
        & (price_sqft > 0)
    )
    in_window["_valid_price_sqft"] = valid_price_sqft

    monthly_records = []
    month_states = []
    coverage_flags = []

    for month in EXPECTED_MONTHS:
        rows = in_window.loc[in_window["_month"] == month]
        valid_ranges = rows.loc[rows["_valid_price_range"]]

        observed_count = int(len(rows))
        sale_count = int(rows["_is_sale"].sum())
        valid_range_count = int(rows["_valid_price_range"].sum())
        valid_sqft_count = int(rows["_valid_price_sqft"].sum())

        if valid_range_count:
            median_min = float(valid_ranges["_min_price_numeric"].median())
            median_max = float(valid_ranges["_max_price_numeric"].median())
        else:
            median_min = None
            median_max = None

        is_partial_month = month == "2023-08"

        if observed_count == 0:
            month_status = "no_listings"
            coverage_status = "no_listings"
            metric_status = "no_listings"
        elif sale_count == 0:
            month_status = "no_valid_prices"
            coverage_status = (
                "partial_month" if is_partial_month else "observed"
            )
            metric_status = "no_sale_listings"
        elif valid_range_count == 0:
            month_status = "no_valid_prices"
            coverage_status = (
                "partial_month" if is_partial_month else "observed"
            )
            metric_status = "no_valid_prices"
        else:
            month_status = "partial_month" if is_partial_month else "ok"
            coverage_status = (
                "partial_month" if is_partial_month else "observed"
            )
            # PRICE_SQFT semantics remain unresolved.
            metric_status = "unverified_metric"

        month_state = MonthState(
            month=month,
            status=month_status,
            observed_listing_count=observed_count,
            sale_listing_count=sale_count,
            valid_price_count=valid_range_count,
            median_min_price_inr=median_min,
            median_max_price_inr=median_max,
            coverage_status=coverage_status,
            is_partial_month=is_partial_month,
            valid_price_sqft_count=valid_sqft_count,
            median_price_per_sqft_inr=None,
        )
        month_state.validate()
        month_states.append(month_state)

        coverage_flag = CoverageFlag(
            month=month,
            observed_listing_count=observed_count,
            sale_listing_count=sale_count,
            valid_price_count=valid_range_count,
            coverage_status=coverage_status,
            is_partial_month=is_partial_month,
            has_observed_listings=observed_count > 0,
            has_valid_prices=valid_range_count > 0,
        )
        coverage_flag.validate()
        coverage_flags.append(coverage_flag)

        monthly_records.append({
            "month": month,
            "observed_listing_count": observed_count,
            "sale_listing_count": sale_count,
            "valid_price_range_count": valid_range_count,
            "valid_price_sqft_count": valid_sqft_count,
            "median_min_price_inr": median_min,
            "median_max_price_inr": median_max,
            "median_price_per_sqft_inr": None,
            "coverage_status": coverage_status,
            "metric_status": metric_status,
        })

    # Whole-source quality accounting.
    valid_dated = int(valid_date_mask.sum())
    in_window_count = int(len(in_window))
    outside_window_count = int(valid_dated - in_window_count)
    in_window_sales = int(in_window["_is_sale"].sum())
    in_window_valid_ranges = int(in_window["_valid_price_range"].sum())

    exact_duplicates = int(df.drop(
        columns=["_posting_date_local", "_month"],
        errors="ignore",
    ).duplicated().sum())

    prop_counts = df["PROP_ID"].dropna().value_counts()
    repeated_groups = int((prop_counts > 1).sum())
    repeated_rows = int(prop_counts[prop_counts > 1].sum())

    data_quality = {
        "raw_rows": int(raw_rows),
        "raw_columns": int(raw_columns),
        "valid_posting_dates": valid_dated,
        "missing_or_invalid_posting_dates": int(raw_rows - valid_dated),
        "valid_dated_rows_outside_window": outside_window_count,
        "in_window_observed_rows": in_window_count,
        "in_window_sale_rows": in_window_sales,
        "in_window_valid_price_ranges": in_window_valid_ranges,
        "in_window_valid_price_sqft_numeric_values": int(
            in_window["_valid_price_sqft"].sum()
        ),
        "in_window_sale_rows_without_valid_coherent_range": (
            in_window_sales - in_window_valid_ranges
        ),
        "exact_duplicate_rows_detected": exact_duplicates,
        "exact_duplicate_rows_removed": 0,
        "repeated_prop_id_groups": repeated_groups,
        "repeated_prop_id_rows": repeated_rows,
    }

    # Reconciliation checks from the global/golden contracts.
    assert valid_dated + data_quality[
        "missing_or_invalid_posting_dates"
    ] == raw_rows
    assert outside_window_count + in_window_count == valid_dated
    assert sum(x["observed_listing_count"] for x in monthly_records) == in_window_count
    assert sum(x["sale_listing_count"] for x in monthly_records) == in_window_sales
    assert sum(x["valid_price_range_count"] for x in monthly_records) == in_window_valid_ranges

    covered_months = sum(
        record["valid_price_range_count"] > 0
        and record["median_min_price_inr"] is not None
        and record["median_max_price_inr"] is not None
        for record in monthly_records
    )
    coverage_score = covered_months / 12.0

    if coverage_score == 1.0:
        overall_coverage_decision = "accept"
        overall_status = "ok"
    elif coverage_score > 0:
        overall_coverage_decision = "partial"
        overall_status = "partial"
    else:
        overall_coverage_decision = "fail"
        overall_status = "partial"

    request = AnalysisRequest()
    plan = RunPlan(
        source_period=SourcePeriod(),
        requested_filters=request.requested_filters,
        effective_filters=request.requested_filters,
        query_mode=request.query_mode,
    )

    runtime_state = RuntimeState(
        run_id="hyderabad-2022-09_2023-08-9e816983",
        request=request,
        plan=plan,
        retries=RetryCounters(),
        coverage_flags=coverage_flags,
        months=month_states,
        trace=[
            {
                "i": 0,
                "agent": "planner",
                "kind": "step",
                "loop": None,
                "msg": "validated default structured request",
                "data": {},
            },
            {
                "i": 1,
                "agent": "scanner",
                "kind": "step",
                "loop": None,
                "msg": "verified and scanned frozen snapshot",
                "data": {
                    "raw_rows": raw_rows,
                    "in_window_rows": in_window_count,
                },
            },
            {
                "i": 2,
                "agent": "aggregator",
                "kind": "step",
                "loop": None,
                "msg": "computed 12 monthly records",
                "data": {"month_count": len(monthly_records)},
            },
            {
                "i": 3,
                "agent": "validator",
                "kind": "decision",
                "loop": None,
                "msg": "checked monthly price-bound coverage",
                "data": {
                    "coverage_score": coverage_score,
                    "coverage_decision": overall_coverage_decision,
                },
            },
        ],
        source_sha256=source_hash,
    )
    runtime_state.validate()

    analysis_report = runtime_state.to_analysis_report(
        data_quality=data_quality,
        seed=0,
        llm_mode="disabled",
        logic_version="hyderabad-runtime-v1",
        status=overall_status,
        coverage_score=coverage_score,
        coverage_decision=overall_coverage_decision,
    )
    report = analysis_report.to_dict()
    report["runtime_state"] = runtime_state.to_dict()

    OUTPUT_FILE.parent.mkdir(parents=True, exist_ok=True)
    OUTPUT_FILE.write_text(
        json.dumps(report, indent=2, allow_nan=False),
        encoding="utf-8",
    )

    print("Runtime state built successfully.")
    print(f"Raw rows: {raw_rows:,}; columns: {raw_columns}")
    print(f"Valid posting dates: {valid_dated:,}")
    print(f"In-window observed rows: {in_window_count:,}")
    print(f"In-window sale rows: {in_window_sales:,}")
    print(f"Valid coherent sale-price ranges: {in_window_valid_ranges:,}")
    print(f"Covered months: {covered_months}/12")
    print(f"Coverage score: {coverage_score:.6f}")
    print(f"Overall status: {overall_status}")
    print(f"Report: {OUTPUT_FILE.relative_to(ROOT)}")


if __name__ == "__main__":
    main()
